export function densityToBodyFat(
  density: number, formula: Formula
): number {
  switch (formula) {
    case 'siri':   bf = 495 / density - 450;   break;
    case 'brozek': bf = 457 / density - 414.2; break;
    case 'behnke': bf = 498 / density - 450;   break;
  }
  return Math.max(2, Math.min(60, bf));
}
